6.BUSINESS INSIGHTS & ANALYSIS

SELECT
round(COUNT(CASE WHEN Status LIKE '%Cancelled%' THEN 1 END)
*100.0/
COUNT(*),1) CancellationRate
FROM Amazon_sale


WITH CategoryRevenue AS
(
SELECT
Category,
SUM(Amount) Revenue
FROM Amazon_sale
GROUP BY Category
),
RevenueRank AS
(
SELECT
Category,
Revenue,
SUM(Revenue) OVER(ORDER BY Revenue DESC) RunningRevenue,
SUM(Revenue) OVER() TotalRevenue
FROM CategoryRevenue
)
SELECT
Category,
Revenue,
ROUND(RunningRevenue*100.0/TotalRevenue,2) CumulativePercent,
CASE
WHEN RunningRevenue*1.0/TotalRevenue<=0.80 THEN 'A Tier'
WHEN RunningRevenue*1.0/TotalRevenue<=0.70 THEN 'B Tier'
ELSE 'C Tier'
END ABC_Class
FROM RevenueRank
ORDER BY Revenue DESC


WITH total_revenue AS
(
SELECT
Category,
SUM(Amount) Revenue
FROM Amazon_sale
GROUP BY Category
),
Pareto AS
(
SELECT *,
SUM(Revenue) OVER(ORDER BY Revenue DESC) RunningRevenue,
SUM(Revenue) OVER() TotalRevenue
FROM total_revenue
)
SELECT *
FROM Pareto
WHERE RunningRevenue<=TotalRevenue*0.80

with month_on_month_rev as (
select month(Date_of_Order) purchase_month,
sum(amount) total_rev
from Amazon_sale
group by month(Date_of_Order))

select*,lag(total_rev) over (order by purchase_month) previous_month_rev,
round((total_rev-lag(total_rev) over (order by purchase_month))*100.0/lag(total_rev) over (order by purchase_month),2)
growth_percent
from month_on_month_rev

WITH MonthlyRevenue AS
(
SELECT
YEAR(Date_of_Order) Yr,
MONTH(Date_of_Order) Mn,
SUM(Amount) Revenue
FROM Amazon_sale
GROUP BY
YEAR(Date_of_Order),
MONTH(Date_of_Order)
)


SELECT
*,
ROUND(
(
Revenue-
LAG(Revenue) OVER(ORDER BY Yr,Mn)
)
*100.0/
NULLIF(LAG(Revenue) OVER(ORDER BY Yr,Mn),0)
,2) GrowthPercent
FROM MonthlyRevenue
